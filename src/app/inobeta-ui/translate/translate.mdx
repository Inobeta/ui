# Translations

<Meta title="Features/Translations/Usage" />

## Usage

```typescript
import { HttpClient } from "@angular/common/http";
import { IbTranslateModuleLoader, IB_TRANSLATE_ASSETS_PATH } from "@inobeta/ui";
import { TranslateLoader, TranslateModule } from "@ngx-translate/core";

export const appConfig: ApplicationConfig = {
  providers: [{
    // this injection token is optional. Default value: './assets/i18n/'
    { provide: IB_TRANSLATE_ASSETS_PATH, useValue: '/public/shared/i18n/' },
    importProvidersFrom([
      TranslateModule.forRoot({
        loader: {
          provide: TranslateLoader,
          useExisting: IbTranslateModuleLoader,
          deps: [HttpClient],
        },
      })
      ]
    ),
  }]
  /** ... */
}
```
